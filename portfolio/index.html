HASHTABLE
hashtable/

Progressive Resizing — two HTable instances for incremental rehashing. Avoids full-stop rehash freezes. FNV hash function with avalanche mixing.

struct HNode {
    HNode *next = NULL;
    uint64_t hcode = 0; // hash value of each node
};

struct HTable {
    HNode **tab = NULL;  // array of buckets

};

struct HMap {
    HTable newer;
    HTable older;

AVL TREE
AVLtrees/
INSERT O(log N) SEARCH O(log N) RANK O(log N)
Zero-Allocation Intrusive BST — embeds AVLNode into ZNode via container_of macro. Branchless pointer-to-pointer rewiring. Identity-theft deletion strategy.
INSPECT SOURCE
struct AVLNode {
    AVLNode *parent = NULL;
    AVLNode *left   = NULL;
    AVLNode *right  = NULL;
    uint32_t height = 0; // height of subtree
    uint32_t cnt    = 0; // subtree size for rank
};

inline void avlInit(AVLNode *node) {
    node->left = node->right = node->parent = NULL;
    node->height = 1;
    node->cnt = 1;
}

AVLNode *avlFix(AVLNode *node);    // rebalance up to root
AVLNode *avlDel(AVLNode *node);    // detach + rebalance
AVLNode *avl_offset(AVLNode *node, int64_t offset);
int64_t  avlRank(AVLNode *node); // 1-based rank

THREAD POOL
threadPool/
QUEUE O(1) DISPATCH ASYNC SHUTDOWN GRACEFUL
Pthread-based async workers — offloads large ZSet deallocations (like Redis UNLINK). Mutex + condition variable signalling. Graceful shutdown flag.

struct Work {
    void (*f)(void *) = NULL; // generic function ptr

struct ThreadPool {
    std::vector<pthread_t> threads;

DLIST (CONN TIMER)
list/
INSERT O(1) REMOVE O(1) IDLE CHECK O(1)
Intrusive circular doubly-linked list — embedded in Conn struct for O(1) idle timeout management. Dummy sentinel node pattern avoids NULL checks.
INSPECT SOURCE
struct DList {
    DList *prev = NULL;
    DList *next = NULL;
};

inline void dlistDetach(DList *node) {
    node->prev->next = node->next;
    node->next->prev = node->prev;
}

inline void dlistInit(DList *node) {
    node->prev = node->next = node; // circular sentinel
}

inline bool dlistEmpty(DList *node) {
    return node->next == node;
}

inline void dlistInsertBefore(DList *target, DList *newNode) {
    DList *prev = target->prev;
    prev->next = newNode;
    newNode->prev = prev;
    newNode->next = target;
    target->prev = newNode;
}
